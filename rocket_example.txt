const output = document.querySelector(".output");
output.textContent = "";
let i = 10;

while (i >= 0 ){
  
  const para = document.createElement('p');
  if (i===10){
    para.textContent = `countdown ${i}`;
    output.appendChild(para);
  }
  else if(i===0) {
    para.textContent = `Blastoff ${i}!!!`;
    output.appendChild(para);
  }
  else {
    para.textContent = i;
    output.appendChild(para);
  }
  
i--;    
}
